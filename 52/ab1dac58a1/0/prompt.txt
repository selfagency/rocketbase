   - **app/src/routes/admin/pages/slug-utils.ts** (Created)
     - Slug generation utility using stopword removal and radashi
     ```typescript
     import { removeStopwords } from 'stopword';
     import { snake } from 'radashi';
     
     export function generateSlug(title: string): string {
       if (!title) return '';
       const words = title.toLowerCase()
         .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
         .replace(/[^\w\s-]/g, '').split(/\s+/).filter(Boolean);
       const wordsWithoutStopwords = removeStopwords(words);
       const finalWords = wordsWithoutStopwords.length > 0 ? wordsWithoutStopwords : words;
       return snake(finalWords.join(' '));
     }
     ```
